1. Variables and Data Types

Python is dynamically typed, meaning you don't need to declare variable types explicitly. Let's explore the basic data types: 

# Numbers
age = 25                    # Integer
price = 19.99              # Float
complex_num = 3 + 4j       # Complex number

# Strings
name = "Alice"
message = 'Hello, World!'
multiline = """This is a
multiline string"""

# Boolean
is_student = True
has_license = False

# None type
result = None

# Type checking
print(type(age))           # <class 'int'>
print(type(price))         # <class 'float'>
print(type(name))          # <class 'str'>

💡 Pro Tip: Use descriptive variable names like student_count instead of sc for better code readability.

2. Control Flow

If-Else Statements

score = 85

if score >= 90:
    grade = "A"
elif score >= 80:
    grade = "B"
elif score >= 70:
    grade = "C"
else:
    grade = "F"

print(f"Your grade is: {grade}")  # Output: Your grade is: B

Loops

# For loop
fruits = ["apple", "banana", "cherry"]
for fruit in fruits:
    print(fruit)

# Range-based loop
for i in range(5):
    print(i)  # Prints 0, 1, 2, 3, 4

# While loop
count = 0
while count < 5:
    print(count)
    count += 1

# List comprehension (Pythonic way)
squares = [x**2 for x in range(10)]
print(squares)  # [0, 1, 4, 9, 16, 25, 36, 49, 64, 81]

3. Data Structures

Lists

# Creating and manipulating lists
numbers = [1, 2, 3, 4, 5]
numbers.append(6)          # Add element
numbers.insert(0, 0)       # Insert at index
numbers.remove(3)          # Remove element
popped = numbers.pop()     # Remove and return last element

# Slicing
print(numbers[1:4])        # [1, 2, 4]
print(numbers[:3])         # [0, 1, 2]
print(numbers[-2:])        # [4, 5]

# List methods
numbers.sort()             # Sort in place
numbers.reverse()          # Reverse in place
print(len(numbers))        # Get length

Dictionaries

# Key-value pairs
student = {
    "name": "Alice",
    "age": 20,
    "courses": ["Math", "Physics", "CS"]
}

# Accessing values
print(student["name"])              # Alice
print(student.get("grade", "N/A"))  # N/A (default value)

# Adding/updating
student["grade"] = "A"
student["age"] = 21

# Dictionary methods
print(student.keys())               # dict_keys(['name', 'age', 'courses', 'grade'])
print(student.values())             # dict_values(['Alice', 21, ['Math', 'Physics', 'CS'], 'A'])
print(student.items())              # Key-value pairs

# Looping through dictionary
for key, value in student.items():
    print(f"{key}: {value}")

Tuples and Sets

# Tuples (immutable)
coordinates = (10, 20)
rgb = (255, 128, 0)
# coordinates[0] = 15  # This would raise an error

# Sets (unique elements)
unique_numbers = {1, 2, 3, 3, 4, 5}
print(unique_numbers)  # {1, 2, 3, 4, 5}

# Set operations
set_a = {1, 2, 3, 4}
set_b = {3, 4, 5, 6}
print(set_a.union(set_b))         # {1, 2, 3, 4, 5, 6}
print(set_a.intersection(set_b))  # {3, 4}
print(set_a.difference(set_b))    # {1, 2}

4. Functions

# Basic function
def greet(name):
    return f"Hello, {name}!"

print(greet("Alice"))  # Hello, Alice!

# Default parameters
def power(base, exponent=2):
    return base ** exponent

print(power(3))      # 9
print(power(3, 3))   # 27

# Multiple return values
def get_stats(numbers):
    return min(numbers), max(numbers), sum(numbers) / len(numbers)

minimum, maximum, average = get_stats([1, 2, 3, 4, 5])

# Lambda functions (anonymous)
square = lambda x: x ** 2
print(square(5))  # 25

# Map, filter, reduce
numbers = [1, 2, 3, 4, 5]
squared = list(map(lambda x: x**2, numbers))
evens = list(filter(lambda x: x % 2 == 0, numbers))

5. Object-Oriented Programming

# Defining a class
class Student:
    # Class variable
    school_name = "Python Academy"
    
    # Constructor
    def __init__(self, name, age):
        self.name = name      # Instance variable
        self.age = age
        self.grades = []
    
    # Instance method
    def add_grade(self, grade):
        self.grades.append(grade)
    
    def get_average(self):
        if not self.grades:
            return 0
        return sum(self.grades) / len(self.grades)
    
    # String representation
    def __str__(self):
        return f"Student: {self.name}, Age: {self.age}"

# Creating objects
alice = Student("Alice", 20)
alice.add_grade(85)
alice.add_grade(90)
alice.add_grade(88)

print(alice)                    # Student: Alice, Age: 20
print(alice.get_average())      # 87.66666666666667
print(Student.school_name)      # Python Academy

# Inheritance
class GraduateStudent(Student):
    def __init__(self, name, age, thesis_topic):
        super().__init__(name, age)
        self.thesis_topic = thesis_topic
    
    def __str__(self):
        return f"Graduate Student: {self.name}, Thesis: {self.thesis_topic}"

bob = GraduateStudent("Bob", 24, "Machine Learning")
print(bob)  # Graduate Student: Bob, Thesis: Machine Learning
