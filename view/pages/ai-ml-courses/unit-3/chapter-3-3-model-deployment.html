1. Model Serialization

Save and load trained models for deployment.

import pickle
import joblib
from sklearn.ensemble import RandomForestClassifier
from sklearn.datasets import load_iris
import tensorflow as tf

# Scikit-learn model
iris = load_iris()
model = RandomForestClassifier()
model.fit(iris.data, iris.target)

# Method 1: Pickle
with open('model.pkl', 'wb') as f:
    pickle.dump(model, f)

# Load with pickle
with open('model.pkl', 'rb') as f:
    loaded_model = pickle.load(f)

# Method 2: Joblib (better for large models)
joblib.dump(model, 'model.joblib')
loaded_model = joblib.load('model.joblib')

# TensorFlow/Keras model
tf_model = tf.keras.Sequential([
    tf.keras.layers.Dense(64, activation='relu', input_shape=(4,)),
    tf.keras.layers.Dense(3, activation='softmax')
])
tf_model.compile(optimizer='adam', loss='sparse_categorical_crossentropy')
tf_model.fit(iris.data, iris.target, epochs=10, verbose=0)

# Save TensorFlow model
tf_model.save('tf_model.h5')  # HDF5 format
tf_model.save('tf_model_saved')  # SavedModel format

# Load TensorFlow model
loaded_tf_model = tf.keras.models.load_model('tf_model.h5')

print("Models saved and loaded successfully!")

2. Flask API

Create a REST API to serve predictions.

from flask import Flask, request, jsonify
import joblib
import numpy as np

app = Flask(__name__)

# Load model at startup
model = joblib.load('model.joblib')

@app.route('/predict', methods=['POST'])
def predict():
    try:
        # Get data from request
        data = request.get_json()
        features = np.array(data['features']).reshape(1, -1)
        
        # Make prediction
        prediction = model.predict(features)
        probability = model.predict_proba(features)
        
        # Return response
        return jsonify({
            'prediction': int(prediction[0]),
            'probability': probability[0].tolist(),
            'status': 'success'
        })
    except Exception as e:
        return jsonify({
            'error': str(e),
            'status': 'error'
        }), 400

@app.route('/health', methods=['GET'])
def health():
    return jsonify({'status': 'healthy'})

if __name__ == '__main__':
    app.run(debug=True, host='0.0.0.0', port=5000)

# Test the API:
# curl -X POST http://localhost:5000/predict \
#   -H "Content-Type: application/json" \
#   -d '{"features": [5.1, 3.5, 1.4, 0.2]}'

3. FastAPI (Modern Alternative)

from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
import joblib
import numpy as np
import uvicorn

app = FastAPI(title="ML Model API", version="1.0")

# Load model
model = joblib.load('model.joblib')

# Define request/response models
class PredictionRequest(BaseModel):
    features: list[float]

class PredictionResponse(BaseModel):
    prediction: int
    probability: list[float]
    status: str

@app.post("/predict", response_model=PredictionResponse)
async def predict(request: PredictionRequest):
    try:
        features = np.array(request.features).reshape(1, -1)
        prediction = model.predict(features)
        probability = model.predict_proba(features)
        
        return PredictionResponse(
            prediction=int(prediction[0]),
            probability=probability[0].tolist(),
            status="success"
        )
    except Exception as e:
        raise HTTPException(status_code=400, detail=str(e))

@app.get("/health")
async def health():
    return {"status": "healthy"}

if __name__ == "__main__":
    uvicorn.run(app, host="0.0.0.0", port=8000)

# Run with: uvicorn main:app --reload
# Docs available at: http://localhost:8000/docs

4. Docker Containerization

Dockerfile

# Dockerfile
FROM python:3.9-slim

WORKDIR /app

# Copy requirements
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# Copy application files
COPY . .

# Expose port
EXPOSE 8000

# Run application
CMD ["uvicorn", "main:app", "--host", "0.0.0.0", "--port", "8000"]

5. Cloud Deployment

AWS Deployment (EC2)

# 1. Launch EC2 instance
# 2. SSH into instance
ssh -i your-key.pem ubuntu@your-ec2-ip

# 3. Install Docker
sudo apt update
sudo apt install docker.io -y
sudo systemctl start docker
sudo systemctl enable docker

# 4. Pull and run your Docker image
sudo docker pull your-dockerhub-username/ml-api:latest
sudo docker run -d -p 80:8000 your-dockerhub-username/ml-api:latest

# 5. Configure security group to allow HTTP traffic on port 80

Google Cloud Run

# 1. Build and push to Google Container Registry
gcloud builds submit --tag gcr.io/PROJECT_ID/ml-api

# 2. Deploy to Cloud Run
gcloud run deploy ml-api \
  --image gcr.io/PROJECT_ID/ml-api \
  --platform managed \
  --region us-central1 \
  --allow-unauthenticated

# 3. Get the service URL
gcloud run services describe ml-api --region us-central1

Heroku Deployment

# 1. Create Procfile
echo "web: uvicorn main:app --host 0.0.0.0 --port $PORT" > Procfile

# 2. Initialize git and create Heroku app
git init
heroku create your-app-name

# 3. Deploy
git add .
git commit -m "Initial deployment"
git push heroku main

# 4. Open app
heroku open

6. Model Monitoring

import logging
from datetime import datetime
import json

# Setup logging
logging.basicConfig(
    filename='model_predictions.log',
    level=logging.INFO,
    format='%(asctime)s - %(message)s'
)

@app.post("/predict")
async def predict(request: PredictionRequest):
    start_time = datetime.now()
    
    try:
        features = np.array(request.features).reshape(1, -1)
        prediction = model.predict(features)
        probability = model.predict_proba(features)
        
        # Log prediction
        log_data = {
            'timestamp': start_time.isoformat(),
            'features': request.features,
            'prediction': int(prediction[0]),
            'probability': probability[0].tolist(),
            'latency_ms': (datetime.now() - start_time).total_seconds() * 1000
        }
        logging.info(json.dumps(log_data))
        
        return PredictionResponse(
            prediction=int(prediction[0]),
            probability=probability[0].tolist(),
            status="success"
        )
    except Exception as e:
        logging.error(f"Prediction error: {str(e)}")
        raise HTTPException(status_code=400, detail=str(e))

# Monitor model performance
@app.get("/metrics")
async def metrics():
    # Read logs and calculate metrics
    with open('model_predictions.log', 'r') as f:
        logs = [json.loads(line.split(' - ')[1]) for line in f.readlines()]
    
    avg_latency = np.mean([log['latency_ms'] for log in logs])
    total_predictions = len(logs)
    
    return {
        'total_predictions': total_predictions,
        'average_latency_ms': avg_latency,
        'status': 'healthy'
    }
