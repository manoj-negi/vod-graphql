Difference between var memo = map[int]int{} vs memo := map[int]int{}

Short Answer

Both create an initialized empty map, but they differ in scope and usage context:

  • var memo = map[int]int{} - Can be used at package level or function level
  • memo := map[int]int{} - Short variable declaration, only works inside functions
package main

import "fmt"

// Package-level variable - MUST use 'var'
var globalMemo = map[int]int{}  // ✅ Valid
// globalMemo2 := map[int]int{} // ❌ Syntax error: cannot use := at package level

func main() {
    // Inside function - both work
    var memo1 = map[int]int{}  // ✅ Valid - explicit declaration
    memo2 := map[int]int{}     // ✅ Valid - short declaration (preferred)
    
    fmt.Println(memo1, memo2)
    
    // Key difference: := infers type and declares
    memo3 := map[string]int{"a": 1}  // Type inferred as map[string]int
    
    // var requires explicit type OR initialization
    var memo4 map[int]int              // Declared but nil (not initialized)
    var memo5 = map[int]int{}          // Declared and initialized
    var memo6 map[int]int = map[int]int{}  // Explicit type and initialization
    
    fmt.Println(memo4 == nil)  // true - nil map
    fmt.Println(memo5 == nil)  // false - initialized empty map
}

💡 Best Practice: Use := inside functions for conciseness. Use var at package level or when you need to declare without initialization.

Array vs Slice: What to Use When?

Key Differences

Array

  • • Fixed size (part of type)
  • • Value type (copied on assignment)
  • • Size known at compile time
  • • Memory allocated on stack (usually)

Slice

  • • Dynamic size (can grow/shrink)
  • • Reference type (shares underlying array)
  • • Size determined at runtime
  • • Points to underlying array on heap
package main

import "fmt"

func main() {
    // ARRAYS - Fixed size
    var arr1 [3]int              // [0, 0, 0]
    arr2 := [3]int{1, 2, 3}      // [1, 2, 3]
    arr3 := [...]int{1, 2, 3, 4} // Size inferred: [4]int
    
    // Arrays are value types - full copy on assignment
    arr4 := arr2
    arr4[0] = 100
    fmt.Println(arr2[0])  // 1 (unchanged)
    fmt.Println(arr4[0])  // 100
    
    // Different sizes = different types
    // var x [3]int = [4]int{1,2,3,4}  // ❌ Type mismatch
    
    // SLICES - Dynamic size
    var slice1 []int                    // nil slice
    slice2 := []int{1, 2, 3}           // Initialized slice
    slice3 := make([]int, 3)           // Length 3, capacity 3
    slice4 := make([]int, 3, 5)        // Length 3, capacity 5
    
    // Slices are reference types - share underlying array
    slice5 := slice2
    slice5[0] = 100
    fmt.Println(slice2[0])  // 100 (changed!)
    fmt.Println(slice5[0])  // 100
    
    // Slices can grow
    slice2 = append(slice2, 4, 5, 6)
    fmt.Println(slice2)  // [100, 2, 3, 4, 5, 6]
    
    // Slicing creates a view (shares memory)
    subSlice := slice2[1:4]  // [2, 3, 4]
    subSlice[0] = 999
    fmt.Println(slice2)      // [100, 999, 3, 4, 5, 6]
}

// Arrays as function parameters - COPIED
func modifyArray(arr [3]int) {
    arr[0] = 100  // Only modifies local copy
}

// Slices as function parameters - SHARED
func modifySlice(s []int) {
    s[0] = 100  // Modifies original slice's underlying array
}

💡 When to Use:

  • Arrays: Fixed-size data (e.g., RGB colors [3]uint8, coordinates [2]float64), performance-critical code needing stack allocation
  • Slices: 99% of cases - dynamic collections, function parameters, working with variable-length data

Pointers in Go & Struct Pointers

package main

import "fmt"

type Person struct {
    Name string
    Age  int
}

func main() {
    // Basic pointers
    x := 42
    p := &x        // p is pointer to x
    fmt.Println(*p) // 42 - dereference to get value
    *p = 100       // Modify value through pointer
    fmt.Println(x)  // 100
    
    // Struct pointers
    person1 := Person{Name: "Alice", Age: 30}
    person2 := &Person{Name: "Bob", Age: 25}  // Returns pointer
    
    // Go automatically dereferences struct pointers
    fmt.Println(person2.Name)   // Syntactic sugar for (*person2).Name
    person2.Age = 26            // Automatically dereferenced
    
    // Passing by value vs pointer
    modifyByValue(person1)
    fmt.Println(person1.Age)    // 30 (unchanged)
    
    modifyByPointer(&person1)
    fmt.Println(person1.Age)    // 31 (changed!)
    
    modifyByPointer(person2)    // Already a pointer
    fmt.Println(person2.Age)    // 27
    
    // nil pointer
    var p3 *Person
    fmt.Println(p3 == nil)      // true
    // fmt.Println(p3.Name)     // ❌ Panic: nil pointer dereference
    
    // new() allocates and returns pointer
    p4 := new(Person)           // *Person with zero values
    p4.Name = "Charlie"
    
    // Method receivers
    person1.Birthday()          // Value receiver - makes copy
    person2.BirthdayPointer()   // Pointer receiver - modifies original
}

// Value receiver - receives copy
func (p Person) Birthday() {
    p.Age++  // Only modifies copy
}

// Pointer receiver - receives pointer
func (p *Person) BirthdayPointer() {
    p.Age++  // Modifies original
}

func modifyByValue(p Person) {
    p.Age++  // Only modifies local copy
}

func modifyByPointer(p *Person) {
    p.Age++  // Modifies original struct
}

💡 When to Use Pointers:

  • 1. Need to modify the original value
  • 2. Large structs (avoid copying overhead)
  • 3. Implementing interfaces that require pointer receivers
  • 4. Sharing state across functions

Empty Struct struct{} & Empty Interface interface{}

package main

import (
    "fmt"
    "unsafe"
)

func main() {
    // EMPTY STRUCT - struct{}
    // Size: 0 bytes - takes NO memory!
    var empty struct{}
    fmt.Println(unsafe.Sizeof(empty))  // 0
    
    // Use case 1: Set implementation (only care about keys)
    set := make(map[string]struct{})
    set["apple"] = struct{}{}
    set["banana"] = struct{}{}
    
    if _, exists := set["apple"]; exists {
        fmt.Println("Apple exists in set")
    }
    
    // Use case 2: Signal-only channels
    done := make(chan struct{})
    go func() {
        // Do work...
        done <- struct{}{}  // Signal completion (no data needed)
    }()
    <-done  // Wait for signal
    
    // Use case 3: Method receiver when no state needed
    type Worker struct{}
    w := Worker{}
    w.DoWork()
    
    // EMPTY INTERFACE - interface{}
    // Can hold ANY type (like Object in Java, void* in C)
    var any interface{}
    
    any = 42
    fmt.Println(any)  // 42
    
    any = "hello"
    fmt.Println(any)  // hello
    
    any = []int{1, 2, 3}
    fmt.Println(any)  // [1 2 3]
    
    // Type assertion to get concrete type
    str, ok := any.(string)
    if ok {
        fmt.Println("String:", str)
    }
    
    // Type switch
    switch v := any.(type) {
    case int:
        fmt.Println("Integer:", v)
    case string:
        fmt.Println("String:", v)
    case []int:
        fmt.Println("Int slice:", v)
    default:
        fmt.Println("Unknown type")
    }
    
    // Modern Go (1.18+): use 'any' alias
    var modern any  // Equivalent to interface{}
    modern = 100
}

type Worker struct{}

func (w Worker) DoWork() {
    fmt.Println("Working...")
}

💡 Key Differences:

  • struct{}: Zero-size type, used for signaling/sets where value doesn't matter
  • interface{} (any): Can hold any type, used for generic containers or when type is unknown

What is make() Used For?

make() is used to initialize slices, maps, and channels (reference types). It allocates and initializes the underlying data structure.

package main

import "fmt"

func main() {
    // ============ SLICES ============
    
    // Without make - nil slice
    var s1 []int
    fmt.Println(s1 == nil)        // true
    fmt.Println(len(s1), cap(s1)) // 0 0
    // s1[0] = 1                  // ❌ Panic: index out of range
    s1 = append(s1, 1)            // ✅ Works - append handles nil slices
    
    // With make - initialized slice
    s2 := make([]int, 3)          // Length 3, capacity 3
    fmt.Println(s2 == nil)        // false
    fmt.Println(len(s2), cap(s2)) // 3 3
    fmt.Println(s2)               // [0 0 0]
    s2[0] = 100                   // ✅ Works
    
    s3 := make([]int, 3, 5)       // Length 3, capacity 5
    fmt.Println(len(s3), cap(s3)) // 3 5
    
    // ============ MAPS ============
    
    // Without make - nil map
    var m1 map[string]int
    fmt.Println(m1 == nil)        // true
    // m1["key"] = 1              // ❌ Panic: assignment to nil map
    val, ok := m1["key"]          // ✅ Reading from nil map is safe
    fmt.Println(val, ok)          // 0 false
    
    // With make - initialized map
    m2 := make(map[string]int)
    fmt.Println(m2 == nil)        // false
    m2["key"] = 1                 // ✅ Works
    fmt.Println(m2["key"])        // 1
    
    // With capacity hint (optimization)
    m3 := make(map[string]int, 100)  // Pre-allocate for ~100 elements
    
    // ============ CHANNELS ============
    
    // Without make - nil channel
    var ch1 chan int
    fmt.Println(ch1 == nil)       // true
    // ch1 <- 1                   // ❌ Deadlock: send on nil channel
    // <-ch1                      // ❌ Deadlock: receive on nil channel
    
    // With make - unbuffered channel
    ch2 := make(chan int)
    go func() {
        ch2 <- 42                 // Send
    }()
    val = <-ch2                   // Receive
    fmt.Println(val)              // 42
    
    // With make - buffered channel
    ch3 := make(chan int, 3)      // Buffer size 3
    ch3 <- 1                      // Non-blocking (buffer not full)
    ch3 <- 2
    ch3 <- 3
    // ch3 <- 4                   // Would block (buffer full)
    fmt.Println(<-ch3)            // 1
    
    // ============ COMPARISON ============
    
    // Literal initialization (alternative to make for slices/maps)
    s4 := []int{1, 2, 3}          // Same as make + assignment
    m4 := map[string]int{"a": 1}  // Same as make + assignment
    
    // new() vs make()
    p1 := new([]int)              // Returns *[]int (pointer to nil slice)
    fmt.Println(*p1 == nil)       // true
    
    p2 := make([]int, 3)          // Returns []int (initialized slice)
    fmt.Println(p2 == nil)        // false
}

💡 Summary:

  • make(): Initializes slices, maps, channels (returns ready-to-use value)
  • new(): Allocates memory, returns pointer to zero value (rarely used)
  • Literal {}: Creates and initializes in one step

Channels: Types & Panic Conditions

package main

import (
    "fmt"
    "time"
)

func main() {
    // ============ CHANNEL TYPES ============
    
    // 1. Unbuffered channel (synchronous)
    ch1 := make(chan int)
    go func() {
        ch1 <- 42  // Blocks until receiver is ready
    }()
    val := <-ch1   // Blocks until sender sends
    fmt.Println(val)
    
    // 2. Buffered channel (asynchronous up to buffer size)
    ch2 := make(chan int, 2)
    ch2 <- 1       // Non-blocking (buffer has space)
    ch2 <- 2       // Non-blocking (buffer has space)
    // ch2 <- 3    // Would block (buffer full)
    fmt.Println(<-ch2, <-ch2)
    
    // 3. Directional channels
    // Send-only channel
    var sendOnly chan<- int = make(chan int)
    go func(ch chan<- int) {
        ch <- 100
        // val := <-ch  // ❌ Compile error: receive from send-only
    }(sendOnly)
    
    // Receive-only channel
    var recvOnly <-chan int = make(chan int)
    go func(ch <-chan int) {
        val := <-ch
        fmt.Println(val)
        // ch <- 100  // ❌ Compile error: send to receive-only
    }(recvOnly)
    
    // ============ PANIC CONDITIONS ============
    
    // 1. Close already closed channel
    ch3 := make(chan int)
    close(ch3)
    // close(ch3)  // ❌ PANIC: close of closed channel
    
    // 2. Send on closed channel
    ch4 := make(chan int, 1)
    close(ch4)
    // ch4 <- 1    // ❌ PANIC: send on closed channel
    
    // 3. Receive from closed channel (NO PANIC - returns zero value)
    ch5 := make(chan int, 2)
    ch5 <- 1
    ch5 <- 2
    close(ch5)
    fmt.Println(<-ch5)  // 1
    fmt.Println(<-ch5)  // 2
    fmt.Println(<-ch5)  // 0 (zero value, no panic)
    
    // Check if channel is closed
    ch6 := make(chan int, 1)
    ch6 <- 42
    close(ch6)
    
    val, ok := <-ch6
    fmt.Println(val, ok)  // 42 true (value was in buffer)
    
    val, ok = <-ch6
    fmt.Println(val, ok)  // 0 false (channel closed, no more values)
    
    // ============ CHANNEL PATTERNS ============
    
    // Pattern 1: Range over channel
    ch7 := make(chan int, 3)
    ch7 <- 1
    ch7 <- 2
    ch7 <- 3
    close(ch7)  // Must close for range to exit
    
    for val := range ch7 {
        fmt.Println(val)  // 1, 2, 3
    }
    
    // Pattern 2: Select with multiple channels
    ch8 := make(chan int)
    ch9 := make(chan string)
    
    go func() {
        time.Sleep(100 * time.Millisecond)
        ch8 <- 42
    }()
    
    go func() {
        time.Sleep(50 * time.Millisecond)
        ch9 <- "hello"
    }()
    
    select {
    case v := <-ch8:
        fmt.Println("Received int:", v)
    case v := <-ch9:
        fmt.Println("Received string:", v)  // This executes first
    case <-time.After(200 * time.Millisecond):
        fmt.Println("Timeout")
    }
    
    // Pattern 3: Non-blocking send/receive
    ch10 := make(chan int, 1)
    
    select {
    case ch10 <- 42:
        fmt.Println("Sent successfully")
    default:
        fmt.Println("Channel full, skipped")
    }
    
    select {
    case val := <-ch10:
        fmt.Println("Received:", val)
    default:
        fmt.Println("No value available")
    }
}

// ============ DEADLOCK CONDITIONS ============

func deadlockExamples() {
    // Deadlock 1: Send on unbuffered channel with no receiver
    ch := make(chan int)
    // ch <- 1  // ❌ DEADLOCK: all goroutines are asleep
    
    // Deadlock 2: Receive on unbuffered channel with no sender
    // val := <-ch  // ❌ DEADLOCK
    
    // Deadlock 3: Send on nil channel
    var nilCh chan int
    // nilCh <- 1  // ❌ DEADLOCK: send on nil channel blocks forever
    
    // Deadlock 4: Receive on nil channel
    // <-nilCh  // ❌ DEADLOCK: receive on nil channel blocks forever
    
    _ = ch
}

⚠️ Panic Conditions:

  • 1. Closing an already closed channel
  • 2. Sending to a closed channel
  • 3. Closing a nil channel

💡 Best Practices:

  • • Only the sender should close channels
  • • Use buffered channels to prevent blocking
  • • Check channel status with val, ok := <-ch
  • • Use select for non-blocking operations

WaitGroup & Wait Conditions

package main

import (
    "fmt"
    "sync"
    "time"
)

func main() {
    // ============ BASIC WAITGROUP ============
    
    var wg sync.WaitGroup
    
    // Add 3 goroutines to wait for
    wg.Add(3)
    
    go worker(1, &wg)
    go worker(2, &wg)
    go worker(3, &wg)
    
    wg.Wait()  // Block until all Done() called
    fmt.Println("All workers completed")
    
    // ============ COMMON PATTERNS ============
    
    // Pattern 1: Dynamic number of goroutines
    tasks := []int{1, 2, 3, 4, 5}
    var wg2 sync.WaitGroup
    
    for _, task := range tasks {
        wg2.Add(1)  // Increment counter
        go func(id int) {
            defer wg2.Done()  // Decrement when done
            processTask(id)
        }(task)  // Pass task to avoid closure issue
    }
    
    wg2.Wait()
    fmt.Println("All tasks processed")
    
    // Pattern 2: WaitGroup with error handling
    var wg3 sync.WaitGroup
    errChan := make(chan error, 3)
    
    for i := 1; i <= 3; i++ {
        wg3.Add(1)
        go func(id int) {
            defer wg3.Done()
            if err := riskyOperation(id); err != nil {
                errChan <- err
            }
        }(i)
    }
    
    wg3.Wait()
    close(errChan)
    
    for err := range errChan {
        fmt.Println("Error:", err)
    }
    
    // Pattern 3: Semaphore with WaitGroup
    maxConcurrent := 2
    sem := make(chan struct{}, maxConcurrent)
    var wg4 sync.WaitGroup
    
    for i := 1; i <= 10; i++ {
        wg4.Add(1)
        go func(id int) {
            defer wg4.Done()
            sem <- struct{}{}        // Acquire semaphore
            defer func() { <-sem }() // Release semaphore
            
            heavyTask(id)
        }(i)
    }
    
    wg4.Wait()
    fmt.Println("All heavy tasks completed")
}

func worker(id int, wg *sync.WaitGroup) {
    defer wg.Done()  // Decrement counter when function returns
    
    fmt.Printf("Worker %d starting\n", id)
    time.Sleep(time.Second)
    fmt.Printf("Worker %d done\n", id)
}

func processTask(id int) {
    fmt.Printf("Processing task %d\n", id)
    time.Sleep(100 * time.Millisecond)
}

func riskyOperation(id int) error {
    time.Sleep(50 * time.Millisecond)
    if id == 2 {
        return fmt.Errorf("operation %d failed", id)
    }
    return nil
}

func heavyTask(id int) {
    fmt.Printf("Heavy task %d running\n", id)
    time.Sleep(500 * time.Millisecond)
}

// ============ COMMON MISTAKES ============

func mistakes() {
    // Mistake 1: Forgetting to call Done()
    var wg sync.WaitGroup
    wg.Add(1)
    go func() {
        // wg.Done()  // ❌ Forgot to call - deadlock!
        time.Sleep(time.Second)
    }()
    // wg.Wait()  // Would wait forever
    
    // Mistake 2: Calling Add() inside goroutine (race condition)
    var wg2 sync.WaitGroup
    for i := 0; i < 5; i++ {
        go func() {
            wg2.Add(1)  // ❌ Race: might call Wait() before Add()
            defer wg2.Done()
        }()
    }
    // wg2.Wait()  // Might return before all goroutines start
    
    // Mistake 3: Reusing WaitGroup without resetting
    var wg3 sync.WaitGroup
    wg3.Add(1)
    go func() { defer wg3.Done() }()
    wg3.Wait()
    
    // wg3.Add(1)  // ✅ Can reuse after Wait() returns
    
    // Mistake 4: Negative counter
    var wg4 sync.WaitGroup
    wg4.Add(1)
    wg4.Done()
    // wg4.Done()  // ❌ PANIC: negative WaitGroup counter
    
    // Mistake 5: Copying WaitGroup
    var wg5 sync.WaitGroup
    wg5.Add(1)
    // wgCopy := wg5  // ❌ Don't copy - pass by pointer
    // Use: func(wg *sync.WaitGroup)
}

💡 WaitGroup Rules:

  • 1. Call Add() before starting goroutine
  • 2. Always call Done() (use defer)
  • 3. Pass WaitGroup by pointer (*sync.WaitGroup)
  • 4. Don't call Add() inside goroutine (race condition)
  • 5. Counter must not go negative (panic)
